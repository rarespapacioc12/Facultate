-- SCRIPT CREATE

CREATE TABLE FILIALA(
    cod_filiala NUMBER(5) PRIMARY KEY,
    denumire VARCHAR2(50) NOT NULL,
    data_infiintare DATE NOT NULL,
    cod_localizare NUMBER(5) NOT NULL,
    cod_info_contact NUMBER(5) NOT NULL
);

CREATE TABLE SEDIU_CENTRAL(
    cod_sediu NUMBER(5) PRIMARY KEY,
    nume_cladire VARCHAR2(50) NOT NULL,
    cod_filiala NUMBER(5) NOT NULL
);

CREATE TABLE MEMBRU(
    cod_membru NUMBER(5) PRIMARY KEY,
    nume VARCHAR2(30) NOT NULL,
    prenume VARCHAR2(30) NOT NULL,
    data_aderare DATE NOT NULL,
    status VARCHAR2(20) NOT NULL,
    cod_filiala NUMBER(5) NOT NULL,
    cod_localizare NUMBER(5) NOT NULL,
    cod_info_contact NUMBER(5) NOT NULL
);

CREATE TABLE CAMPANIE(
    cod_campanie NUMBER(5) PRIMARY KEY,
    titlu VARCHAR2(100),
    buget_estimat NUMBER(12,2) NOT NULL,
    data_start DATE NOT NULL,
    data_final DATE NOT NULL
);

CREATE TABLE ALEGERI(
    cod_alegeri NUMBER(5) PRIMARY KEY,
    denumire VARCHAR2(50),
    tip VARCHAR2(30) NOT NULL,
    an NUMBER(4) NOT NULL
);

CREATE TABLE FUNCTIE_ELECTORALA(
    cod_functie NUMBER(5) PRIMARY KEY,
    denumire_functie VARCHAR2(50) NOT NULL,
    indemnizatie_recomandata NUMBER(10, 2)
);

CREATE TABLE DONATOR(
    cod_donator NUMBER(5) PRIMARY KEY,
    nume_donator VARCHAR2(50) NOT NULL,
    tip VARCHAR2(20) NOT NULL,
    cod_localizare NUMBER(5) NOT NULL,
    cod_info_contact NUMBER(5) NOT NULL
);

CREATE TABLE LOCALIZARE(
    cod_localizare NUMBER(5) PRIMARY KEY,
    adresa VARCHAR2(100) NOT NULL,
    oras VARCHAR2(50) NOT NULL,
    judet VARCHAR2(50) NOT NULL,
    tara VARCHAR2(30) DEFAULT 'Romania'
);

CREATE TABLE INFO_CONTACT(
    cod_info_contact NUMBER(5) PRIMARY KEY,
    telefon VARCHAR2(15) NOT NULL,
    mail VARCHAR2(50) NOT NULL,
    fax VARCHAR2(15)
);

CREATE TABLE CATEGORIE_SPECIALA(
    cod_categorie NUMBER(5) PRIMARY KEY,
    nume_categorie VARCHAR2(20) NOT NULL
);

CREATE TABLE CANDIDAT(
    cod_membru NUMBER(5) PRIMARY KEY,
    functie_vizata VARCHAR2(50) NOT NULL,
    experienta_politica_ani NUMBER(2) NOT NULL
); 

CREATE TABLE COTIZATIE(
    cod_membru NUMBER(5),
    nr_chitanta NUMBER(10),
    data_plata DATE NOT NULL,
    suma_plata NUMBER(10, 2) NOT NULL,
    CONSTRAINT pk_cotizatie PRIMARY KEY(cod_membru, nr_chitanta)
);

CREATE TABLE DONATIE(
    cod_donator NUMBER(5),
    cod_donatie NUMBER(10),
    suma NUMBER(15, 2) NOT NULL,
    data_donatie DATE NOT NULL,
    cod_ordin_plata VARCHAR2(20) NOT NULL,
    CONSTRAINT pk_donatie PRIMARY KEY(cod_donator, cod_donatie)
);

CREATE TABLE PARTICIPA(
    cod_campanie NUMBER(5),
    cod_membru NUMBER(5),
    data_inscriere DATE NOT NULL,
    ore_implicare NUMBER(3) NOT NULL,
    CONSTRAINT pk_participa PRIMARY KEY(cod_campanie, cod_membru)
);

CREATE TABLE APARTINE(
    cod_membru NUMBER(5),
    cod_categorie NUMBER(5),
    data_inscriere DATE NOT NULL,
    beneficii VARCHAR2(100),
    CONSTRAINT pk_apartine PRIMARY KEY(cod_membru, cod_categorie)
);

CREATE TABLE CANDIDATURA(
    cod_membru NUMBER(5),
    cod_alegeri NUMBER(5),
    cod_functie NUMBER(5),
    buget NUMBER(10, 2) NOT NULL,
    CONSTRAINT pk_candidatura PRIMARY KEY(cod_membru, cod_alegeri, cod_functie)
);
